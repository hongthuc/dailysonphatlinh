        /* Cơ chế tải ảnh Google Drive thông minh (thử lần lượt các endpoint của Google Drive) */
        function handleDriveImgError(imgEl, labelText) {
            const driveId = imgEl.getAttribute('data-drive-id');
            const step = parseInt(imgEl.getAttribute('data-fallback-step') || '0', 10);

            if (driveId && step === 0) {
                imgEl.setAttribute('data-fallback-step', '1');
                imgEl.src = 'https://lh3.googleusercontent.com/d/' + driveId + '=w800';
                return;
            }
            if (driveId && step === 1) {
                imgEl.setAttribute('data-fallback-step', '2');
                imgEl.src = 'https://drive.google.com/uc?export=view&id=' + driveId;
                return;
            }
            imgEl.onerror = null;
            const encoded = encodeURIComponent(labelText || 'Sơn Phát Linh');
            imgEl.src = 'https://placehold.co/500x500/002D62/FFFFFF?text=' + encoded;
        }

            const filtered = catSlug === 'all'
                ? productsData
                : productsData.filter(item => item.category === catSlug);
            renderAllProducts(filtered, categoryTitles[catSlug] || 'DANH SÁCH SẢN PHẨM');
        }

        function filterByBrand(brandName) {
            const filtered = productsData.filter(item => item.brand.toLowerCase() === brandName.toLowerCase());
            renderAllProducts(filtered, 'SẢN PHẨM CHÍNH HÃNG SƠN ' + brandName.toUpperCase());
        }

        /* Tìm kiếm sản phẩm */
        function executeSearch(keyword, catSlug) {
            const cleanKey = (keyword || '').trim().toLowerCase();
            let list = productsData;
            if (catSlug && catSlug !== 'all') {
                list = list.filter(p => p.category === catSlug);
            }
            if (cleanKey) {
                list = list.filter(p =>
                    p.name.toLowerCase().includes(cleanKey) ||
                    p.categoryLabel.toLowerCase().includes(cleanKey) ||
                    p.brand.toLowerCase().includes(cleanKey)
                );
            }
            renderAllProducts(list, cleanKey ? `KẾT QUẢ TÌM KIẾM: "${keyword}"` : categoryTitles[catSlug || 'all']);
            document.getElementById('danh-muc-san-pham').scrollIntoView({ behavior: 'smooth' });
        }

        /* Điều khiển 2 Banner tự trượt chuyên nghiệp */
        let currentBannerIndex = 0;
        let bannerInterval = null;

        function showBannerSlide(index) {
            const slides = document.querySelectorAll('.banner-slide');
            const dot0 = document.getElementById('bannerDot0');
            const dot1 = document.getElementById('bannerDot1');
            if (!slides.length) return;

            currentBannerIndex = (index + slides.length) % slides.length;
            slides.forEach((sl, idx) => {
                if (idx === currentBannerIndex) {
                    sl.classList.add('active');
                } else {
                    sl.classList.remove('active');
                }
            });

            if (dot0 && dot1) {
                if (currentBannerIndex === 0) {
                    dot0.className = 'w-6 h-2 rounded-full bg-brand-amber transition-all';
                    dot1.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all';
                } else {
                    dot1.className = 'w-6 h-2 rounded-full bg-brand-amber transition-all';
                    dot0.className = 'w-2.5 h-2 rounded-full bg-white/50 transition-all';
                }
            }
        }

        function changeBannerSlide(step) {
            showBannerSlide(currentBannerIndex + step);
            restartBannerAutoSlide();
        }

        function goToBannerSlide(index) {
            showBannerSlide(index);
            restartBannerAutoSlide();
        }

        function restartBannerAutoSlide() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                showBannerSlide(currentBannerIndex + 1);
            }, 5000);
        }
